 deploy-docker-image-from-docker-hub-to-AWS-EC2:
    name: Deploy to AWS EC2
    needs:
      - fetch_aws_ec2_instance_id
    runs-on: ubuntu-latest

    permissions:
      id-token: write
      contents: read
      issues: write
      pull-requests: write

    # Job-level environment variables
    env:
      AWS_REGION: ${{ secrets.AWS_REGION }}
      EC2_INSTANCE_ID: ${{ needs.fetch_aws_ec2_instance_id.outputs.EC2_INSTANCE_ID }}
      DOCKER_USER: ${{ secrets.DOCKER_USERNAME }}

    steps:
      - name: Checkout Repository
        uses: actions/checkout@v4

      - name: Configure AWS Credentials via OIDC
        uses: aws-actions/configure-aws-credentials@v4
        with:
          role-to-assume: ${{ secrets.AWS_ROLE }}
          aws-region: ${{ env.AWS_REGION }}

      - name: Verify EC2 Instance ID
        run: |
          echo "Deploying to Target Instance ID: $EC2_INSTANCE_ID"

      - name: Execute Remote Deployment via SSM
        env:
          # Map GitHub Secrets or Workflow Variables here
          
        run: |
            set -e
            
            PUBLIC_IP=$(aws ec2 describe-instances \
              --instance-ids "$EC2_INSTANCE_ID" \
              --query "Reservations[0].Instances[0].PublicIpAddress" \
              --output text)
            
            echo "Deploying application to target IP: https://$PUBLIC_IP"
    
            # 1. Define docker-compose YAML content
            COMPOSE_YAML=$(cat <<EOF
            
            EOF
            )
    
            # 2. Define .env content
            ENV_FILE_CONTENT=$(cat <<EOF
            
            EOF
            )
    
            # 3. Base64 encode both payloads
            B64_COMPOSE=$(echo "$COMPOSE_YAML" | base64 -w 0)
            B64_ENV=$(echo "$ENV_FILE_CONTENT" | base64 -w 0)
    
            # 4. Construct JSON payload for SSM (with error redirection and Docker V2 syntax)
            COMMANDS_JSON=$(jq -n \
              --arg b64_compose "$B64_COMPOSE" \
              --arg b64_env "$B64_ENV" \
              '{commands: [
                "mkdir -p /home/ubuntu/cicd_kc_admin_console",
                "echo \"\($b64_compose)\" | base64 -d > /home/ubuntu/cicd_kc_admin_console/docker-compose.yml",
                "echo \"\($b64_env)\" | base64 -d > /home/ubuntu/cicd_kc_admin_console/.env",
                "echo \"--- Inspecting Decoded Files ---\"",
                "cat /home/ubuntu/cicd_kc_admin_console/docker-compose.yml",
                "echo \"--- Pulling Latest Docker Image ---\"",
                "sudo docker compose -f /home/ubuntu/cicd_kc_admin_console/docker-compose.yml pull 2>&1 || sudo docker-compose -f /home/ubuntu/cicd_kc_admin_console/docker-compose.yml pull 2>&1",
                "echo \"--- Starting Docker Compose ---\"",
                "sudo docker compose -f /home/ubuntu/cicd_kc_admin_console/docker-compose.yml up -d --remove-orphans 2>&1 || sudo docker-compose -f /home/ubuntu/cicd_kc_admin_console/docker-compose.yml up -d --remove-orphans 2>&1",
                "sleep 3",
                "echo \"--- Container Verification ---\"",
                "sudo docker ps -a"
            ]}')
    
            # 5. Send SSM Command
            CMD_ID=$(aws ssm send-command \
              --document-name "AWS-RunShellScript" \
              --instance-ids "$EC2_INSTANCE_ID" \
              --parameters "$COMMANDS_JSON" \
              --output text \
              --query "Command.CommandId")
    
            echo "Sent SSM Command ID: $CMD_ID"
    
            # 6. Polling Loop
            for i in $(seq 1 36); do
              STATUS=$(aws ssm get-command-invocation \
                --command-id "$CMD_ID" \
                --instance-id "$EC2_INSTANCE_ID" \
                --query "Status" \
                --output text 2>/dev/null || echo "Pending")
    
              echo "Deployment Status ($i/36): $STATUS"
              [ "$STATUS" = "Success" ] && break
              [[ "$STATUS" == "Failed" || "$STATUS" == "TimedOut" || "$STATUS" == "Cancelled" ]] && exit 1
              sleep 10
            done
    
            # 7. Output Execution Logs
            echo "--- EC2 Execution Logs ---"
            aws ssm get-command-invocation \
              --command-id "$CMD_ID" \
              --instance-id "$EC2_INSTANCE_ID" \
              --query "StandardOutputContent" \
              --output text
